Gramática:

Valor ->  Table(Objeto) 
        | Array
        | String
        | Número
        | Bool
        | Null


Table ->  [Titulo]\nAtributo
        | Texto = { Atributo }

Titulo ->  Texto
         | Texto.Titulo

Atributo ->  Texto = Valor
           | Texto = Valor\nAtributo
           | Texto.Atributo

Texto ->  [a-z]
        | [a-z]Texto



Array ->  [ Elemento ]
        | [ ]

Elemento ->  Valor
           | Valor, Elemento



String ->  Basico
         | MultiLineBasic
         | Literal
         | MultiLineLiteral

Basico ->  " Texto "

MultiLineBasic -> """ Texto """

Literal -> ' Texto '

MultiLineLiteral -> ''' Texto '''



Número ->  SinalTipos
         | Tipos

Sinal ->  +
        | -

Tipos ->  Digito
        | Digito_Tipos
        | Digito.Float

Float ->  Digito
        | DigitoFloat
        | SinalFloat
        | (e|E) Float

Digito ->  [ 0, 9 ]
         | [ 1, 9 ]Digito



Bool ->  true
       | false
